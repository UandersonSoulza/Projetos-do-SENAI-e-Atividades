CREATE DATABASE sistvendas;
USE sistvendas;

SELECT*FROM sistvendas;

CREATE TABLE CLIENTES (
  CPF VARCHAR(14) BINARY  NOT NULL   AUTO_INCREMENT,
  NOME VARCHAR(255)  NOT NULL  ,
  dateNASCIMENTO DATE  NOT NULL  ,
  SEXO VARCHAR(20)  NOT NULL    ,
PRIMARY KEY(CPF));

CREATE TABLE COMPRAS (
  ID_REGISTRO INTEGER UNSIGNED  NOT NULL   AUTO_INCREMENT,
  CLIENTES_CPF VARCHAR(14) BINARY  NOT NULL  ,
  PRODUTOS_ID_PRODUTOS INTEGER UNSIGNED  NOT NULL    ,
PRIMARY KEY(ID_REGISTRO, CLIENTES_CPF, PRODUTOS_ID_PRODUTOS)  ,
INDEX COMPRAS_FKIndex1(CLIENTES_CPF)  ,
INDEX COMPRAS_FKIndex2(PRODUTOS_ID_PRODUTOS));

CREATE TABLE PRODUTOS (
  ID_PRODUTOS INTEGER UNSIGNED  NOT NULL   AUTO_INCREMENT,
  DATA_VALIDADE DATE  NOT NULL  ,
  NOME INTEGER UNSIGNED  NOT NULL  ,
  MARCA INTEGER UNSIGNED  NOT NULL  ,
  PRECO FLOAT(10,2)  NOT NULL    ,
PRIMARY KEY(ID_PRODUTOS));
